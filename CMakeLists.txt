cmake_minimum_required(VERSION 3.16)
project(ColorSearch LANGUAGES CXX)
add_executable(ColorSearch WIN32 src/main.cpp)
target_compile_features(ColorSearch PRIVATE cxx_std_17)
target_link_libraries(ColorSearch PRIVATE gdiplus comdlg32 gdi32 ole32)
if(MINGW)
  target_compile_options(ColorSearch PRIVATE -finput-charset=UTF-8)
  target_link_options(ColorSearch PRIVATE -municode -static)
endif()
if(MSVC)
  target_compile_options(ColorSearch PRIVATE /utf-8)
endif()
